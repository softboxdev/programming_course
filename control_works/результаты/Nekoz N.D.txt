import sqlite3

def delete_task_by_id(task_id):
    conn = sqlite3.connect('app.db')
    cur = conn.cursor()
    try:
        cur.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
        conn.commit()
        return cur.rowcount
    except:
        conn.rollback()
        return 0
    finally:
        conn.close()
